func (this *cTrace) _begin(name *string, printAll bool) {
	this._levelPlus()
	this.getActiveLevelItem().funcName = *name
	var content, precFuncNames string
	if printAll {
		this.getContent(&content)
	}
	if this.tracePrecs {
		this.preparePrec(&precFuncNames)
	}
	fmt.Fprintf(this.fd, "%s>> %s%s %s\n", this.tab(), traceStyleBegin.Echo(*name), content, precFuncNames)
}

func (this *cTrace) getContent(s *string) {
	active := this.getActiveLevelItem()
	*s = active.content
	active.content = ""
}

func (this *cTrace) _beginEnd(name *string) {
	this._levelPlus()
	// this.funcNames[this.level] = *name
	var content, precFuncNames string
	this.getContent(&content)
	// this.getContent(&content)
	if this.tracePrecs {
		this.preparePrec(&precFuncNames)
	}
	fmt.Fprintf(this.fd, "%s<> %s%s %s\n", this.tab(), traceStyleBegin.Echo(*name), content, precFuncNames)
	this._levelMinus()
}

func (this *cTrace) _end() {
	var content, s string
	var style *cTraceStyle
	active := this.getActiveLevelItem()
	this.getContent(&content)
	if active.isSilent {
		s = "<>"
		style = &traceStyleBegin
		active.isSilent = false
	} else {
		s = "<<"
		style = &traceStyleEnd
	}
	fmt.Fprintf(this.fd, "%s%s %s%s\n", this.tab(), s, style.Echo(active.funcName), content)
	this._levelMinus()
}

///

func (this *cTrace) _return() {
	var content, s string
	var style *cTraceStyle
	active := this.getActiveLevelItem()
	this.getContent(&content)
	if active.isSilent {
		s = "<>"
		style = &traceStyleBegin
		active.isSilent = false
	} else {
		s = "<<"
		style = &traceStyleEnd
	}
	fmt.Fprintf(this.fd, "%s%s return %s%s\n", this.tab(), s, style.Echo(active.funcName), content)
	this._levelMinus()
}

func (this *cTrace) ReturnAdd(format string, a ...any) {
	this.AddToPrint(format, a...)
	this._return()
}

func (this *cTrace) Return() {
	this._return()
}

///

func (this *cTrace) AddToPrint(format string, a ...any) {
	this.getActiveLevelItem().content += fmt.Sprintf(" · "+format, a...)
}

func (this *cTrace) AddToPrintPlus1(format string, a ...any) {
	var level = this.level + 1
	if level >= len(this.levelItems) {
		level--
	}
	this.levelItems[level].content += fmt.Sprintf(" · "+format, a...)
}

func (this *cTrace) Error_n(t iTrace, format string, a ...any) {
	fo := fmt.Sprintf("%s!! %s.%s\n",
		this.getActiveLevelItem().tab,
		t.GetLongName(),
		format)
	fmt.Fprintf(this.fd, traceStyleError.Echo(fo), a...)
}

func (this *cTrace) Error(format string, a ...any) {
	fo := fmt.Sprintf("%s!! %s\n",
		this.getActiveLevelItem().tab,
		format)
	fo2 := fmt.Sprintf("%s\n", format)
	fmt.Fprintf(this.fd, traceStyleError.Echo(fo), a...)
	var precs string
	this.preparePrec2(&precs)
	fmt.Fprintln(this.fdErr, precs)
	fmt.Fprintf(this.fdErr, fo2, a...)
	fmt.Fprintln(this.fdErr)
}

func (this *cTrace) Err(err error) {
	fmt.Fprintf(this.fd, "err: %v\n", err)
}

// Give a slice of iTrace, get list of names.
func GetLongNames[T iTrace](names []T) string {
	var s string
	for _, name := range names {
		s += name.GetLongName() + ", "
	}
	return s
}

func TraceNameAndInterfaceFromMap[T iTrace](ma map[string]T) {
	for name, iface := range ma {
		trace.Print("   %-20s %s", name, iface.GetLongName())
	}
}

// Prints an indented "%#v".
func (this *cTrace) PrintIndentedV(a ...any) {
	var v string
	var level int
	var tab = "    "
	var Tabs = func() string {
		return strings.Repeat(tab, level)
	}
	// var TabsMinus = func() string {
	// 	if level > 0 {
	// 		level--
	// 	}
	// 	return strings.Repeat(tab, level)
	// }
	for i := range a {
		aa := a[i]
		v += fmt.Sprintf("---- %#v\n", aa)
	}
	// v = strings.ReplaceAll(v, ", ", ",\n    ")
	v = strings.ReplaceAll(v, "{", "{\n\t")
	v = strings.ReplaceAll(v, "}", "}\n\t")
	v = strings.ReplaceAll(v, ", ", ",\n\t")
	// v = strings.ReplaceAll(v, "nil", DC_BOLD+"<NIL>"+DC_RESET+traceStyleNormal.JustTheColor())
	v = strings.ReplaceAll(v, "nil", traceStyleNormal.Bold("<NIL>"))
	vv := strings.Split(v, "\t")
	for _, v := range vv {
		fmt.Fprint(this.fd, traceStyleNormal.Echo(Tabs()+v))
		if strings.HasSuffix(v, "{\n") {
			level++
		} else if strings.HasSuffix(v, "}\n") {
			level--
		}
	}
	fmt.Println()
}

func (this *cTrace) _levelPlus() {
	this.level++
	if this.level >= len(this.levelItems) {
		this.level = len(this.levelItems) - 1
	}
}

func (this *cTrace) preparePrec(precFuncNames *string) {
	*precFuncNames = "("
	for i := range this.level {
		*precFuncNames += this.levelItems[i].funcName + " >> "
	}
	*precFuncNames += ")"
	*precFuncNames = DC_RESET + traceStyleNormal.Echo(*precFuncNames)
}

func (this *cTrace) preparePrec2(precFuncNames *string) {
	for i := range this.level {
		*precFuncNames += this.levelItems[i+1].funcName + " >> "
	}
}

func (this *cTrace) _levelMinus() {
	this.level--
	if this.level < 0 {
		this.level = 0
		trace.Error("TRACE --- level < 1")
	}
}

func GetOuterNilOrName(outer iGetName) string {
	if outer == nil {
		return "<nil>"
	} else {
		return outer.GetName()
	}
}
