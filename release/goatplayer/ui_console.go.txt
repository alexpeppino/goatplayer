func RemoteInterface_go() {
	trace.GoPrint(">>>> RemoteInterface START")
	var send string
	// Create a Unix domain socket and listen for incoming connections.
	os.Remove("/tmp/echo.sock")
	socket, err := net.Listen("unix", "/tmp/echo.sock")
	if err != nil {
		trace.Error("%v", err)
		log.Print(err)
		return
	}
	// rmPrintCommandList()
	for {
		/// Accept an incoming connection.
		conn, err := socket.Accept()
		if err != nil {
			log.Fatal(err)
		}
		/// Create a buffer for incoming data.
		buf := make([]byte, 4096)
		// Read data from the connection.
		trace.GoPrint(">>>> RemoteInterface READ")
		n, err := conn.Read(buf)
		if err != nil {
			log.Fatal(err)
		}
		s := string(buf[:n-1])
		if s != "x" {
			RemConsole.rmArgs = strings.Split(s, " ")
		}
		trace.GoPrint(">>>> RemoteInterface ExecCommand: %v", RemConsole.rmArgs)
		if cmm, ok := RemConsole.mapCmm[RemConsole.rmArgs[0]]; ok {
			send = ""
			cmm.f()
		} else {
			trace.GoPrint(">>>> RemoteInterface invalid command")
		}
		bs := []byte(send + "\n")
		/// Echo the data back to the connection.
		// s := []byte("come stai?\n")
		_, err = conn.Write(bs)
		// _, err = conn.Write(buf[:n])
		if err != nil {
			trace.GoPrint(">>>> error writing to pipe, %v", err)
		}
		conn.Close()
	}
}
