func NameOrNil(a iTrace, s string) {
	if a == nil {
		trace.Print("%s: nil", s)
	} else {
		trace.Print("%s: %s", s, a.GetLongName())
	}
}

func (cDump) TraceTerm() {
	var tab = "   "
	trace.Print("-- TraceTerm --")
	trace.Print("MainScreen.pSectionRight: %s", MainScreen.pRightSection.name)
	trace.Print("activeScreen: %s", Term.activeScreen.name)

		trace.Print("oldActiveScreen: %s", Term.oldActiveScreen.name)
	}

		trace.Print("activeScreen.activeWindow: %s", Term.activeScreen.activeWindow.name)

		trace.Print("Screen.pSectionRight.activeWindow: %s", MainScreen.pRightSection.GetActiveWindow().name)

		MainScreen.pRightSection.GetActiveWindow().lines.TraceSelected(tab)
	} else {
		trace.Print("Screen.pSectionRight.activeLayer.activeWindow: NIL")
	}
	trace.Print("")
	trace.Print("")

func (this *vMainScreen) TraceActives() {
	var s string
	s = "activeSection"
	if this.activeSection == nil {
		trace.Print("%s: nil", s)
	} else {
		trace.Print("%s: %s", s, this.activeSection.GetLongName())
		s = "activeSection.actLayer"
		if this.activeSection.GetActiveLayer() == nil {
			trace.Print("%s: nil", s)
		} else {
			trace.Print("%s: %s", s, this.activeSection.GetActiveLayer().GetLongName())
			s = "activeSection.actLayer.activeWindow"
			if this.activeSection.GetActiveLayer().activeWindow == nil {
				trace.Print("%s: nil", s)
			} else {
				trace.Print("%s: %s", s, this.activeSection.GetActiveLayer().activeWindow.GetLongName())
			}
		}
	}
	s = "activeWindow"
	if this.activeWindow == nil {
		trace.Print("%s: nil", s)
	} else {
		trace.Print("%s: %s", s, this.activeWindow.GetLongName())
	}
}

func (this *vScreen) TraceScreenFocuses() {
	trace.Begin_n(this, "TraceScreenFocuses")
	focus := this.outer.GetFocusMan_o()

		trace.Print("   key / model:  %6q  %s", Input.GetKeyName(key), model.name)
	}

func (cDump) TraceAndReportError(comment string, doReport bool) {
	s1 := fmt.Sprintf("Error: %s (%s)", errorBuffer.String(), comment)
	s1 = strings.ReplaceAll(s1, "\n", "")
	trace.Error(s1)
	if doReport {
		Report.AddErrorLine(s1)
	}
	errorBuffer.Reset()
}
